async function generar(){ if(!validarPaso(4))return; const datos=recolectarDatos(); document.getElementById('form-wrap').style.display='none'; document.getElementById('panel-generando').style.display='block'; window.scrollTo(0,0); const steps=document.querySelectorAll('.as-gen-step'); let si=0; const stepTimer=setInterval(()=>{ steps.forEach((s,i)=>s.classList.toggle('act',i===si)); si=Math.min(si+1,steps.length-1); },15000); try{ const controller=new AbortController(); const timeoutId=setTimeout(()=>controller.abort(),270000); const res=await fetch(`/api/assessment/${UID}`,{ method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify(datos), signal:controller.signal }); clearTimeout(timeoutId); clearInterval(stepTimer); const data=await res.json(); document.getElementById('panel-generando').style.display='none'; document.getElementById('panel-resultado').style.display='block'; if(res.ok && data.estado==='listo'){ document.getElementById('res-icon').textContent='🎉'; document.getElementById('res-title').textContent='¡Tu plan está listo!'; document.getElementById('res-sub').textContent=`${data.ejercicios_cargados} ejercicios cargados`; document.getElementById('res-btn-app').style.display='block'; } else { document.getElementById('res-icon').textContent='⚠'; document.getElementById('res-title').textContent='Error al generar el plan'; document.getElementById('res-sub').textContent=data.error||'Error desconocido'; document.getElementById('res-btn-retry').style.display='block'; } }catch(e){ clearInterval(stepTimer); document.getElementById('panel-generando').style.display='none'; document.getElementById('panel-resultado').style.display='block'; document.getElementById('res-icon').textContent='⚠'; document.getElementById('res-title').textContent='Error de conexión'; document.getElementById('res-sub').textContent=e.name==='AbortError'?'Tiempo agotado. Intenta de nuevo.':e.message; document.getElementById('res-btn-retry').style.display='block'; } }
